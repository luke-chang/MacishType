# The bound comes from jshost/, folded in below; this project's own files ask
# for far less. Declaring it here keeps the real requirement discoverable.
cmake_minimum_required(VERSION 3.25)
project(fcitx5-macishtype VERSION 0.3.0)

include("${CMAKE_CURRENT_SOURCE_DIR}/../shared/cmake/SourcePathMap.cmake")

# Directories CMake creates on the way to a destination otherwise follow the
# build umask, which lands group-writable 775 under /usr/share.
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS
    OWNER_READ OWNER_WRITE OWNER_EXECUTE
    GROUP_READ GROUP_EXECUTE
    WORLD_READ WORLD_EXECUTE)

# The embedded interpreter runs on every keystroke; never build it
# unoptimized by accident.
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# 5.1.13 is the release that added the StandardPaths API engine discovery
# builds on; asking for it here turns an older fcitx5 into a configure-time
# error instead of a wall of missing-header output.
find_package(Fcitx5Core 5.1.13 REQUIRED)
find_package(Fontconfig REQUIRED)
find_package(Fcitx5Module REQUIRED COMPONENTS Notifications)
find_package(Gettext REQUIRED)
include("${FCITX_INSTALL_CMAKECONFIG_DIR}/Fcitx5Utils/Fcitx5CompilerSettings.cmake")

# The shared engine host (jshost/ at the repo root, which owns the quickjs-ng
# submodule and the system-ICU Intl bindings) builds as a static library
# folded into the addon .so, so everything under it needs PIC.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../jshost" jshost EXCLUDE_FROM_ALL)

add_subdirectory(po)
add_subdirectory(src)

# Data tables backing the manifest `modules` and the associated-phrase
# dictionary, plus the corpus licenses that must ship with them — only what
# the host reads (the Array tables in ../tables are engine data, copied by
# engines/array30/setup.py). Generated by the repo-level
# scripts/ExternalResources.py; installed when present.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tables")
    install(FILES
        "${CMAKE_CURRENT_SOURCE_DIR}/../tables/AssociatedDictionary.zh-Hant.txt"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tables/WordFrequency.zh-Hant.txt"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tables/SymbolNames.zh-Hant.txt"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tables/McBopomofo.LICENSE.txt"
        "${CMAKE_CURRENT_SOURCE_DIR}/../tables/CLDR.LICENSE.txt"
        DESTINATION "${FCITX_INSTALL_PKGDATADIR}/macishtype/tables")
endif()

# Engines installed alongside the addon (the Makefile's PRELOAD_ENGINES).
# They land in the system XDG layer, where a user-layer folder of the same
# name still shadows them.
set(MACISHTYPE_PRELOAD_ENGINES "" CACHE STRING
    "Engine ids under ../engines to install with the addon, or `all`")
string(TOLOWER "${MACISHTYPE_PRELOAD_ENGINES}" macishtype_preload_keyword)
if(macishtype_preload_keyword STREQUAL "all")
    # Every folder carrying a manifest, the rule scripts/InstallEngines.py
    # discovers by. Re-evaluated on each configure, which install and pkg
    # always run.
    file(GLOB macishtype_manifests
         "${CMAKE_CURRENT_SOURCE_DIR}/../engines/*/manifest.json")
    set(macishtype_preload "")
    foreach(manifest IN LISTS macishtype_manifests)
        get_filename_component(macishtype_found "${manifest}" DIRECTORY)
        get_filename_component(macishtype_found "${macishtype_found}" NAME)
        # A swept-up folder discovery would skip is simply not ours to
        # install; a named one is a typo worth failing on (below).
        if(macishtype_found MATCHES "^[A-Za-z0-9_-]+$")
            list(APPEND macishtype_preload "${macishtype_found}")
        endif()
    endforeach()
    # What `all` came to depends on the directory, so say it rather than
    # leaving it to be read off the payload.
    list(JOIN macishtype_preload ", " macishtype_all_text)
    message(STATUS "PRELOAD_ENGINES=all: ${macishtype_all_text}")
else()
    string(REPLACE "," " " macishtype_preload "${MACISHTYPE_PRELOAD_ENGINES}")
    separate_arguments(macishtype_preload UNIX_COMMAND "${macishtype_preload}")
endif()
if(macishtype_preload)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
endif()
foreach(engine IN LISTS macishtype_preload)
    # Engine discovery accepts only these characters, so anything else would
    # ship a folder the addon cannot load.
    if(NOT engine MATCHES "^[A-Za-z0-9_-]+$")
        message(FATAL_ERROR "PRELOAD_ENGINES: \"${engine}\" is not a valid"
                            " engine id (want [A-Za-z0-9_-])")
    endif()
    set(engine_dir "${CMAKE_CURRENT_SOURCE_DIR}/../engines/${engine}")
    if(NOT EXISTS "${engine_dir}/manifest.json")
        message(FATAL_ERROR
                "PRELOAD_ENGINES: engines/${engine}/manifest.json not found")
    endif()
    # An engine's generated data tables are gitignored; its setup.py copies
    # them out of ../tables, running the resource pipeline itself when they
    # are missing. Running it here is what leaves the folder complete for the
    # install rule below.
    if(EXISTS "${engine_dir}/setup.py")
        execute_process(COMMAND "${Python3_EXECUTABLE}" setup.py
                        WORKING_DIRECTORY "${engine_dir}"
                        RESULT_VARIABLE macishtype_setup_result)
        if(NOT macishtype_setup_result EQUAL 0)
            message(FATAL_ERROR
                    "PRELOAD_ENGINES: engines/${engine}/setup.py failed")
        endif()
    endif()
    # Only what the engine needs at runtime. setup.py in particular cannot
    # run from an installed copy — it reads the repo's scripts/.
    install(DIRECTORY "${engine_dir}/"
            DESTINATION
                "${FCITX_INSTALL_PKGDATADIR}/macishtype/engines/${engine}"
            PATTERN "setup.py" EXCLUDE
            PATTERN "*.test.mjs" EXCLUDE
            PATTERN "README.md" EXCLUDE
            PATTERN "_storage" EXCLUDE
            PATTERN "__pycache__" EXCLUDE
            PATTERN ".git*" EXCLUDE)
endforeach()

# Debian policy: every package documents its licensing under
# /usr/share/doc/<package>/copyright (DEP-5 format).
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/copyright"
        DESTINATION "share/doc/fcitx5-macishtype")

# Native packages (`make pkg`): CPack assembles them from the install rules
# above. DESTDIR staging keeps the fcitx5-derived absolute destinations
# as-is, so a package carries exactly what `make install` lays down.
set(CPACK_SET_DESTDIR ON)
set(CPACK_STRIP_FILES ON)
# The package version carries the build stamp identifying the commit it was
# built from. `+` keeps the stamp out of precedence decisions and is legal for
# both dpkg and rpm (`^` would suit rpm but dpkg rejects it); it goes in the
# version rather than the packaging release field so `dpkg -l` / `rpm -q`
# still show it after install. The stamp is decided at build time and written
# next to the binary (see src/CMakeLists.txt); reading it back when cpack runs
# keeps package and binary in agreement by construction.
set(CPACK_PROJECT_CONFIG_FILE "${CMAKE_CURRENT_BINARY_DIR}/CPackVersion.cmake")
configure_file(cmake/CPackVersion.cmake.in CPackVersion.cmake @ONLY)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY
    "MacishType JavaScript input method engines for fcitx5")
set(CPACK_PACKAGE_DESCRIPTION
    "Runs input method engines written in JavaScript against the MacishType \
contract on an embedded quickjs-ng runtime. Every engine folder becomes its \
own fcitx5 input method, with hot reload and native settings.")
if(macishtype_preload)
    list(JOIN macishtype_preload ", " macishtype_preload_text)
    string(APPEND CPACK_PACKAGE_DESCRIPTION
           " Preloaded engines: ${macishtype_preload_text}.")
endif()
set(CPACK_PACKAGE_VENDOR "Luke Chang")
set(CPACK_PACKAGE_CONTACT "Luke Chang <luke1209@gmail.com>")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/luke-chang/MacishType")
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
# Library dependencies (ICU, fontconfig, Fcitx5Core, ...) are computed from
# the linked sonames; only the framework and the capability modules the code
# consults are declared by hand. Both generators repeat the find_package
# version bound: an RPM soname dependency carries none, so an older fcitx5
# would install the package and then fail to load the addon.
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "fcitx5 (>= 5.1.13), fcitx5-modules")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS
    "fcitx5-module-chttrans, fcitx5-module-fullwidth, libcanberra0")
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
# CPackRPM reads its own description variable and otherwise falls back to
# CMake's generic boilerplate.
set(CPACK_RPM_PACKAGE_DESCRIPTION "${CPACK_PACKAGE_DESCRIPTION}")
# RPM-family distributions ship the capability modules in one package.
set(CPACK_RPM_PACKAGE_REQUIRES "fcitx5 >= 5.1.13")
set(CPACK_RPM_PACKAGE_RECOMMENDS "fcitx5-chinese-addons libcanberra")
# rpmbuild stamps the build machine's hostname into the header, where the
# source path mapping cannot reach it.
set(CPACK_RPM_SPEC_MORE_DEFINE "%define _buildhost reproducible")
# The install rules carry fcitx5-derived absolute paths, which DESTDIR
# staging cannot turn into a relocatable package. CPack requests one by
# default; opting out here matches what the package actually is.
set(CPACK_PACKAGE_RELOCATABLE OFF)
include(CPack)
